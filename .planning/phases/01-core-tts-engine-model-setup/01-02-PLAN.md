    yield
    # Cleanup if needed

app = FastAPI(lifespan=lifespan)

def get_tts_service(request: Request):
    return request.app.state.tts_service
```

       from contextlib import asynccontextmanager
       from fastapi import FastAPI, Request, Depends, HTTPException
       from fastapi.responses import JSONResponse
       from services.tts import TTSModel
       from models.schemas import VoicesResponse, VoiceInfo, TTSRequest
       import logging

       # Configure logging
       logging.basicConfig(level=logging.INFO)
       logger = logging.getLogger(__name__)

       @asynccontextmanager
       async def lifespan(app: FastAPI):
           """Load TTS model once at startup."""
           logger.info("Loading TTS model...")
           try:
               app.state.tts_service = TTSModel()
               logger.info(f"TTS model loaded successfully on device: {app.state.tts_service.device}")
           except Exception as e:
               logger.error(f"Failed to load TTS model: {e}")
               raise
           yield
           # Cleanup on shutdown
           logger.info("Shutting down TTS service...")

       app = FastAPI(
           title="Local TTS API",
           description="High-quality local text-to-speech synthesis",
           version="1.0.0",
           lifespan=lifespan
       )

       # CORS middleware for local development
       from fastapi.middleware.cors import CORSMiddleware
       app.add_middleware(
           CORSMiddleware,
           allow_origins=["*"],  # Restrict in production
           allow_methods=["*"],
           allow_headers=["*"],
       )

       def get_tts_service(request: Request) -> TTSModel:
           """Dependency injection for TTS service."""
           return request.app.state.tts_service
       ```

    3. Implement POST /tts endpoint:
       ```python
       @app.post("/tts")
       async def generate_tts(
           request: TTSRequest,
           tts: TTSModel = Depends(get_tts_service)
       ):
           """Generate audio from text.
           
           Returns audio as WAV stream.
           """
           try:
               logger.info(f"Generating TTS for voice={request.voice}, text_length={len(request.text)}")
               
               # Validate voice
               if request.voice not in tts.voices:
                   raise HTTPException(status_code=400, detail=f"Invalid voice: {request.voice}")
               
               # Generate audio
               audio = tts.generate(
                   text=request.text,
                   voice=request.voice,
                   speed=request.speed
               )
               
               if audio is None:
                   raise HTTPException(status_code=500, detail="Audio generation failed")
               
               # Convert numpy array to WAV bytes
               buffer = io.BytesIO()
               sf.write(buffer, audio, 24000, format='WAV')
               buffer.seek(0)
               
               return StreamingResponse(
                   buffer,
                   media_type="audio/wav",
                   headers={
                       "X-Voice": request.voice,
                       "X-Text-Length": str(len(request.text))
                   }
               )
               
           except HTTPException:
               raise
           except Exception as e:
               logger.error(f"TTS generation error: {e}")
               raise HTTPException(status_code=500, detail=f"Generation failed: {str(e)}")
       ```
