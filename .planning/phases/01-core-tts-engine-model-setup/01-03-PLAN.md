       """Text preprocessing utilities for TTS."""
       import re
       from typing import List

       MAX_CHUNK_LENGTH = 500  # Characters per chunk for stability
       MAX_INPUT_LENGTH = 5000  # Maximum total input length

       def preprocess_text(text: str) -> str:
           """Clean and normalize text for TTS.
           
           Args:
               text: Raw input text
               
           Returns:
               Cleaned text ready for TTS
           """
           if not text:
               return ""
           
           # Normalize unicode (convert smart quotes, etc.)
           text = text.replace('"', '"').replace('"', '"')
           text = text.replace("'", "'").replace("'", "'")
           text = text.replace('–', '-').replace('—', '-')
           
           # Strip excessive whitespace
           text = " ".join(text.split())
           
           # Remove control characters except newlines
           text = "".join(char for char in text if char == '\n' or ord(char) >= 32)
           
           # Ensure text ends with punctuation for natural ending
           if text and text[-1] not in '.!?':
               text = text + '.'
           
           return text.strip()

       def chunk_text(text: str, max_length: int = MAX_CHUNK_LENGTH) -> List[str]:
           """Split long text into chunks at sentence boundaries.
           
           Args:
               text: Preprocessed text
               max_length: Maximum characters per chunk
               
           Returns:
               List of text chunks
           """
           if len(text) <= max_length:
               return [text]
           
           chunks = []
           # Split on sentence boundaries
           sentences = re.split(r'(?<=[.!?])\s+', text)
           
           current_chunk = ""
           for sentence in sentences:
               if len(current_chunk) + len(sentence) + 1 <= max_length:
                   current_chunk = (current_chunk + " " + sentence).strip() if current_chunk else sentence
               else:
                   if current_chunk:
                       chunks.append(current_chunk)
                   # If single sentence is too long, split on clauses
                   if len(sentence) > max_length:
                       chunks.extend(_split_long_sentence(sentence, max_length))
                   else:
                       current_chunk = sentence
           
           if current_chunk:
               chunks.append(current_chunk)
           
           return chunks if chunks else [text[:max_length]]

       def _split_long_sentence(sentence: str, max_length: int) -> List[str]:
           """Split a very long sentence on commas or clauses."""
           # Split on commas, semicolons, or conjunctions
           clauses = re.split(r'(?<=[,;])\s+|\s+(and|but|or|so)\s+', sentence)
           
           chunks = []
           current = ""
           for clause in clauses:
               if not clause:
                   continue
               if len(current) + len(clause) + 1 <= max_length:
                   current = (current + " " + clause).strip() if current else clause
               else:
                   if current:
                       chunks.append(current)
                   current = clause
           
           if current:
               chunks.append(current)
           
           # If still too long, hard split at word boundaries
           if not chunks:
               words = sentence.split()
               current = ""
               for word in words:
                   if len(current) + len(word) + 1 <= max_length:
                       current = (current + " " + word).strip() if current else word
                   else:
                       chunks.append(current)
                       current = word
               if current:
                   chunks.append(current)
           
           return chunks

       def validate_input(text: str) -> tuple[bool, str]:
           """Validate input text for TTS.
           
           Returns:
               (is_valid, error_message)
           """
           if not text or not text.strip():
               return False, "Text cannot be empty"
           
           if len(text) > MAX_INPUT_LENGTH:
               return False, f"Text too long (max {MAX_INPUT_LENGTH} characters)"
           
           # Check for excessive repeated characters (spam)
           if re.search(r'(.)\1{10,}', text):
               return False, "Text contains excessive repeated characters"
           
           return True, ""
       ```

       import os
       os.environ["PYTORCH_ENABLE_MPS_FALLBACK"] = "1"

       import torch
       import numpy as np
       from kokoro import KPipeline
       from typing import List, Optional
       import logging

       from models.schemas import VoiceInfo
       from utils.text import preprocess_text, chunk_text, validate_input
       import config

       logger = logging.getLogger(__name__)

       class TTSModel:
           def __init__(self):
               self.device = self._get_device()
               logger.info(f"Initializing TTS on device: {self.device}")
               
               self.pipeline = KPipeline(lang_code=config.LANG_CODE)
               self.voices = config.VOICE_CONFIGS
               
               self._warmup()
               logger.info(f"TTS initialized with {len(self.voices)} voices")
           
           def _get_device(self) -> torch.device:
               if torch.backends.mps.is_available():
                   return torch.device("mps")
               return torch.device("cpu")
           
           def _warmup(self):
               """Warm up model with dummy inference."""
               try:
                   list(self.pipeline("Hello", voice=config.DEFAULT_VOICE))
                   logger.info("Model warmup complete")
               except Exception as e:
                   logger.warning(f"Warmup failed (non-critical): {e}")
           
           def get_voices(self) -> List[VoiceInfo]:
               return [VoiceInfo(id=k, name=v) for k, v in self.voices.items()]
           
           def generate(
               self, 
               text: str, 
               voice: str = None, 
               speed: float = 1.0
           ) -> Optional[np.ndarray]:
               """Generate audio from text.
               
               Args:
                   text: Input text (will be preprocessed)
                   voice: Voice ID to use
                   speed: Speech speed multiplier
                   
               Returns:
                   Audio as numpy array or None if failed
               """
               # Validate input
               is_valid, error = validate_input(text)
               if not is_valid:
                   raise ValueError(error)
               
               # Preprocess text
               text = preprocess_text(text)
               if not text:
                   raise ValueError("Text is empty after preprocessing")
               
               # Validate voice
               voice = voice or config.DEFAULT_VOICE
               if voice not in self.voices:
                   logger.warning(f"Invalid voice '{voice}', using default")
                   voice = config.DEFAULT_VOICE
               
               # Handle long text with chunking
               chunks = chunk_text(text)
               logger.info(f"Processing {len(chunks)} chunk(s) for voice={voice}")
               
               if len(chunks) == 1:
                   # Single chunk - simple generation
                   return self._generate_chunk(chunks[0], voice, speed)
               else:
                   # Multiple chunks - concatenate audio
                   audio_chunks = []
                   for i, chunk in enumerate(chunks):
                       logger.debug(f"Generating chunk {i+1}/{len(chunks)}")
                       audio = self._generate_chunk(chunk, voice, speed)
                       if audio is not None:
                           audio_chunks.append(audio)
                   
                   if audio_chunks:
                       return np.concatenate(audio_chunks)
                   return None
           
           def _generate_chunk(
               self, 
               text: str, 
               voice: str, 
               speed: float
           ) -> Optional[np.ndarray]:
               """Generate audio for a single chunk."""
               try:
                   with torch.no_grad():
                       generator = self.pipeline(text, voice=voice, speed=speed)
                       results = list(generator)
                       if results:
                           return results[0].audio
               except Exception as e:
                   logger.error(f"Generation failed for chunk: {e}")
               return None
       ```

       async def generate_tts(
           request: TTSRequest,
           tts: TTSModel = Depends(get_tts_service)
       ):
           try:
               # Validate voice exists
               if request.voice not in tts.voices:
                   raise HTTPException(status_code=400, detail=f"Invalid voice: {request.voice}")
               
               # Generate audio (includes validation and preprocessing)
               audio = tts.generate(
                   text=request.text,
                   voice=request.voice,
                   speed=request.speed
               )

               buffer = io.BytesIO()

               buffer.seek(0)
               
               return StreamingResponse(
                   buffer,

                       "X-Text-Length": str(len(request.text)),
                       "X-Chunk-Count": str(len(chunk_text(request.text)))
                   }
               )
               
           except ValueError as e:
               # Input validation error
               raise HTTPException(status_code=400, detail=str(e))
